Airflow DAG structure (excerpt)

from airflow import DAG
from airflow.sensors.filesystem import FileSensor
from airflow.operators.python import PythonOperator
from datetime import datetime

with DAG(
    "retail_sales_pipeline",
    schedule_interval="@daily",
    start_date=datetime(2026, 1, 1),
    catchup=True,
) as dag:

    wait_for_files = FileSensor(
        task_id="wait_for_daily_files",
        filepath="/data/raw/{{ ds }}/",
        poke_interval=300,
        timeout=3600,
    )

    bronze_task = PythonOperator(
        task_id="load_bronze",
        python_callable=run_bronze_job,
        op_kwargs={"ds": "{{ ds }}"},
    )

    wait_for_files >> bronze_task

PySpark: daily sales aggregation

sales_by_store = (
    silver_df
    .groupBy("sale_date", "store_id", "category")
    .agg(
        F.sum("line_total").alias("total_sales"),
        F.countDistinct("order_id").alias("order_count"),
    )
)

(sales_by_store.write
    .mode("overwrite")
    .partitionBy("sale_date")
    .parquet("s3://retail-pipeline/gold/sales_by_store/"))
