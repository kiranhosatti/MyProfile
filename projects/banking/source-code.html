Lambda: trigger Glue job on S3 upload

import boto3

glue = boto3.client("glue")

def lambda_handler(event, context):
    bucket = event["Records"][0]["s3"]["bucket"]["name"]
    key = event["Records"][0]["s3"]["object"]["key"]

    response = glue.start_job_run(
        JobName="kyc-validation-job",
        Arguments={
            "--input_path": f"s3://{bucket}/{key}",
        },
    )
    return {"jobRunId": response["JobRunId"]}

PySpark: KYC field validation (excerpt)

from pyspark.sql import functions as F

kyc_df = spark.read.parquet("s3://banking-pipeline/bronze/kyc/")

validated_df = kyc_df.withColumn(
    "is_valid",
    F.col("full_name").isNotNull()
    & F.col("date_of_birth").isNotNull()
    & (F.length("id_number") >= 6)
)

flagged_df = validated_df.filter(~F.col("is_valid"))
clean_df = validated_df.filter(F.col("is_valid"))
